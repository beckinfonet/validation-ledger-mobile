    **Change 1: Add `performHeartbeatIfNeeded(container:)` helper method.**

    Place the new helper alongside the existing `presentBiometricLockIfNeeded(container:over:)` private method. Signature and body:

    ```swift
    /// Phase 4 D-07 heartbeat helper. Fire-and-forget: a failed heartbeat does NOT block
    /// role-shell render. Errors log at .error level with safe fields only (status + event name);
    /// attestationObject / attestedKeyId bytes NEVER enter Logger fields (FOUND-01 PII discipline).
    ///
    /// Invocation sites (D-07):
    ///   1. scene(_:willConnectTo:) .restored branch — cold-boot heartbeat
    ///   2. handleDidBecomeActive — warm-foreground 24h check
    ///
    /// 24h threshold: reads container.attestedKeyStore.readLastHeartbeatAt(); if nil or
    /// older than 24h, fires heartbeat. On success, updates container.session.trustTier +
    /// persists lastHeartbeatAt.
    ///
    /// Stale-session edge case (RESEARCH Pitfall 6): if the session token is server-stale,
    /// the heartbeat POST returns 401 → existing Auth401ResponseInterceptor triggers
    /// LogoutService.logout(.auth401). This is NORMAL; heartbeat serves dual duty as session
    /// validity probe.
    @MainActor
    private func performHeartbeatIfNeeded(container: AppContainer) async {
        do {
            let attestedKeyStore = AttestedKeyStore(keychain: container.keychainStore)
            // 24h threshold check
            let lastHeartbeat = try attestedKeyStore.readLastHeartbeatAt()
            let now = Date()
            if let last = lastHeartbeat,
               now.timeIntervalSince(last) < 86400 {  // 24h = 86400s
                return  // No heartbeat needed yet
            }

            // Get the existing attestedKeyId (or attest fresh if missing — shouldn't happen
            // post-register, but handle gracefully)
            let (keyId, status) = try await container.attestationService.generateKeyIfNeeded()
            guard status == .attested || status == .simulatorBypass else {
                // Can't heartbeat without a key — status already routed to softwareOnly path.
                container.logger.warn(event: .init("attestation_heartbeat_skipped_no_key"),
                                      fields: [/* safe fields: status rawValue */])
                return
            }

            // Fetch challenge (D-05)
            let challengeResponse = try await container.apiClient.send(DeviceChallengeEndpoint())
            guard let challengeData = Data(base64Encoded: challengeResponse.challenge) else {
                container.logger.error(event: .init("attestation_heartbeat_bad_challenge"),
                                       fields: [/* safe fields only */])
                return
            }

            // Generate assertion (D-06)
            let assertion = try await container.attestationService.generateAssertion(
                keyId: keyId,
                challenge: challengeData
            )

            // Read session token from Keychain (existing Phase 3 pattern)
            let sessionTokenData = try container.keychainStore.get(.sessionToken)
            guard let sessionToken = String(data: sessionTokenData, encoding: .utf8) else {
                container.logger.error(event: .init("attestation_heartbeat_no_session_token"),
                                       fields: [])
                return
            }

            // POST /device/heartbeat
            let heartbeatResponse = try await container.apiClient.send(DeviceHeartbeatEndpoint(
                sessionToken: sessionToken,
                attestedKeyId: keyId,
                assertion: assertion
            ))

            // Update trustTier (D-12) + persist lastHeartbeatAt
            container.session.trustTier = heartbeatResponse.trustTier
            try attestedKeyStore.writeLastHeartbeatAt(heartbeatResponse.heartbeatAcceptedAt)

            container.logger.info(event: .init("attestation_heartbeat_ok"),
                                  fields: [/* status + trustTier rawValue — safe */])
        } catch {
            // Silent-fail: do not block role-shell render. Backend will drive re-attest
            // on next /device/register if trustTier demands it. 401 errors are handled
            // by the existing Auth401ResponseInterceptor elsewhere.
            container.logger.error(event: .init("attestation_heartbeat_failed"),
                                   fields: [/* safe — String(describing: error) WITHOUT userInfo */])
        }
    }
    ```

    Exact field names (`attestation_heartbeat_ok`, `attestation_heartbeat_failed`, etc.) follow the "attestation_*" prefix convention from 04-PATTERNS.md Pattern A (Open Q2 resolution: reuse `.event` rather than adding LogField cases).
