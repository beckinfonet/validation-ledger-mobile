    ```swift
    // validationLedger/Core/Networking/Mock/MockDefaultFixtures.swift
    // DEBUG-only fixture factory that lets a developer run the app on a physical
    // device with networkConfig == .mock and tap through the organic onboarding
    // flow without passing any launch args or having a backend running.
    //
    // This file is NOT compiled into Release — entire body is wrapped in
    // `#if DEBUG`. Defense-in-depth: the AppContainer call site is ALSO
    // `#if DEBUG` gated, and the MockURLProtocol instance only exists when
    // networkConfig == .mock (which defaultNetworkConfig(env:) forces to .live
    // in Release).
    //
    // Scope (endpoints registered):
    //   1. POST /auth/otp/request  → 200 { otp_session_id: "dev-mock-otp-session" }
    //   2. POST /auth/otp/verify   → accepts ANY 6-digit numeric code; non-6-digit → 400
    //                                 → 200 { session_token: "dev-mock-session-token",
    //                                         role: "carrier", ... }
    //   3. GET  /device/challenge  → 200 { challenge: base64("dev-mock-challenge"),
    //                                       expires_at: now + 5m, nonce: UUID }
    //   4. POST /device/register   → 200 with trust_tier = "softwareOnly" so the
    //                                 Limited-Trust banner (Plan 04-08) is visible
    //                                 organically in the DEBUG walkthrough.
    //   5. POST /device/heartbeat  → 200 { heartbeat_accepted_at: now,
    //                                       trust_tier: "softwareOnly" }
    //
    // Rationale for softwareOnly default:
    //   - Phase 4 ships the D-11 banner behaviour; a developer doing the organic
    //     walkthrough should see it.
    //   - D-10 simulator-bypass + .mock network mode already imply "non-production
    //     trust surface" — softwareOnly is the honest signal.
    //
    // Non-goals:
    //   - Not a replacement for the `-MockOTPRoleForUITest` path (that bypasses
    //     OTP entirely and is exercised by Phase 3 Plan 12 smoke tests).
    //   - Not a test double — the five endpoint tests in Phase 2-4 register their
    //     own fixtures via MockURLProtocol.reset() + explicit registrations.
    //   - Not a security surface — hard-coded session token is labelled as dev-only.

    #if DEBUG
    import Foundation

    public enum MockDefaultFixtures {

        /// Idempotent: calling twice registers the handlers twice. Callers
        /// should guard via a one-shot flag or call MockURLProtocol.reset()
        /// first. AppContainer's call site is gated on construction so the
        /// factory runs exactly once per app session.
        public static func registerAppDefaults() {
            MockURLProtocol.register(Self.handler)
        }

        /// Single dispatch handler — inspects request.url.path and routes to
        /// the appropriate canned response. Unknown paths fall through to
        /// a 404 response so a missing mock is VISIBLE (does not hang silently).
        private static func handler(_ request: URLRequest) -> (HTTPURLResponse, Data) {
            let path = request.url?.path ?? "<nil>"
            let method = request.httpMethod ?? "GET"

            switch (method, path) {
            case ("POST", "/auth/otp/request"):
                return make200(body: otpRequestResponseJSON(), url: request.url)

            case ("POST", "/auth/otp/verify"):
                if isSixDigitBody(request.httpBody) {
                    return make200(body: otpVerifyResponseJSON(), url: request.url)
                } else {
                    return makeError(
                        status: 400,
                        code: "invalid_code",
                        message: "Enter any 6 digits for the dev mock.",
                        url: request.url
                    )
                }

            case ("GET", "/device/challenge"):
                return make200(body: deviceChallengeResponseJSON(), url: request.url)

            case ("POST", "/device/register"):
                return make200(body: deviceRegisterResponseJSON(), url: request.url)

            case ("POST", "/device/heartbeat"):
                return make200(body: deviceHeartbeatResponseJSON(), url: request.url)

            default:
                return makeError(
                    status: 404,
                    code: "no_mock_fixture",
                    message: "MockDefaultFixtures has no handler for \(method) \(path). Register one in MockDefaultFixtures.swift or switch to .live.",
                    url: request.url
                )
            }
        }

        // MARK: - JSON body builders
        // Strings match fixture shapes in validationLedgerTests/Networking/Fixtures/;
        // field names MUST match the endpoint Response CodingKeys. Update carefully
        // if the real endpoints' contracts evolve.

        private static func otpRequestResponseJSON() -> Data {
            Data(#"{"otp_session_id":"dev-mock-otp-session","resend_after_seconds":30}"#.utf8)
        }

        private static func otpVerifyResponseJSON() -> Data {
            Data(#"{"session_token":"dev-mock-session-token","refresh_token":"dev-mock-refresh-token","role":"carrier","user_id":"dev-mock-user"}"#.utf8)
        }

        private static func deviceChallengeResponseJSON() -> Data {
            let nonce = UUID().uuidString
            let expiresAt = ISO8601DateFormatter().string(from: Date().addingTimeInterval(300))
            // base64("dev-mock-challenge") = "ZGV2LW1vY2stY2hhbGxlbmdl"
            return Data(#"{"challenge":"ZGV2LW1vY2stY2hhbGxlbmdl","expires_at":"\#(expiresAt)","nonce":"\#(nonce)"}"#.utf8)
        }

        private static func deviceRegisterResponseJSON() -> Data {
            let registeredAt = ISO8601DateFormatter().string(from: Date())
            return Data(#"{"device_id":"dev-mock-device","registered_at":"\#(registeredAt)","trust_tier":"softwareOnly"}"#.utf8)
        }

        private static func deviceHeartbeatResponseJSON() -> Data {
            let now = ISO8601DateFormatter().string(from: Date())
            return Data(#"{"heartbeat_accepted_at":"\#(now)","trust_tier":"softwareOnly"}"#.utf8)
        }

        // MARK: - Response builders

        private static func make200(body: Data, url: URL?) -> (HTTPURLResponse, Data) {
            let response = HTTPURLResponse(
                url: url ?? URL(string: "https://mock.local/")!,
                statusCode: 200,
                httpVersion: "HTTP/1.1",
                headerFields: ["Content-Type": "application/json"]
            )!
            return (response, body)
        }

        private static func makeError(status: Int, code: String, message: String, url: URL?) -> (HTTPURLResponse, Data) {
            let response = HTTPURLResponse(
                url: url ?? URL(string: "https://mock.local/")!,
                statusCode: status,
                httpVersion: "HTTP/1.1",
                headerFields: ["Content-Type": "application/json"]
            )!
            let body = Data(#"{"error_code":"\#(code)","message":"\#(message)"}"#.utf8)
            return (response, body)
        }

        /// Accepts any numeric 6-character OTP code. The request body shape depends
        /// on OTPVerifyEndpoint.Request; we inspect the JSON body if present.
        private static func isSixDigitBody(_ body: Data?) -> Bool {
            guard let body,
                  let obj = try? JSONSerialization.jsonObject(with: body) as? [String: Any] else {
                return false
            }
            // Accept any string field whose value matches ^[0-9]{6}$
            for (_, v) in obj {
                if let s = v as? String, s.count == 6, s.allSatisfy({ $0.isNumber }) {
                    return true
                }
            }
            return false
        }
    }
    #endif
    ```
