    // Test cleanup: every @Test purges the attestedKeyId + lastHeartbeatAt Keychain entries in
    // a defer block. Avoids state leakage across runs on the same physical iPhone.

    // Apple quota tolerance (RESEARCH Pitfall 2): attestKey / generateAssertion on a real device
    // occasionally returns serverUnavailable (interpreted as quotaExceeded per D-09). The test
    // accepts either:
    //   - (status == .attested)           → happy path, assertion round-trip completes
    //   - (status == .quotaExceeded)      → quota bite, test logs + passes (acceptable CI outcome)
    // A status of .error or .unknownSystemFailure fails the test.

    import Testing
    import Foundation
    import DeviceCheck
    @testable import validationLedger

    @Suite("AppAttest round-trip — D-13 (3) device real-hardware")
    struct AppAttestRoundTripTests {

        private func purgeAttestationKeychain(keychain: KeychainStore) {
            try? keychain.delete(.attestedKeyId)
            try? keychain.delete(.lastHeartbeatAt)
        }

        @Test("DCAppAttestService is supported on this device")
        func attestServiceSupported() {
            #expect(DCAppAttestService.shared.isSupported == true,
                    "Device tests require App Attest capability; pre-A10 or missing iOS 17 would fail here")
        }

        @Test("attestKey returns non-empty attestationObject OR .quotaExceeded (accept either per Pitfall 2)")
        func attestKeyRoundTrip() async throws {
            // Build a keychain store + service under the real device's Secure Enclave.
            let keychain = KeychainStore(service: "com.maldin.validationLedger.tests.attest")
            purgeAttestationKeychain(keychain: keychain)
            defer { purgeAttestationKeychain(keychain: keychain) }

            let attestedKeyStore = AttestedKeyStore(keychain: keychain)
            let logger = OSLogLoggerImpl(subsystem: LoggingSubsystem.auth, category: "auth.attestation.test")
            let service = DCAppAttestAttestationService(
                keyStore: attestedKeyStore,
                logger: logger
            )

            let (keyId, status) = try await service.generateKeyIfNeeded()
            // Accept either .attested or .quotaExceeded
            #expect(status == .attested || status == .quotaExceeded,
                    "D-13-3 + Pitfall 2: status should be .attested (happy) or .quotaExceeded (quota); got \(status)")

            // If we got a keyId, exercise attestKey round-trip
            if status == .attested, !keyId.isEmpty {

                do {
                    let attestationObject = try await service.attestKey(keyId: keyId, challenge: challenge)
                    #expect(!attestationObject.isEmpty,
                            "attestationObject CBOR must be non-empty on happy path")
                } catch AttestationError.rateLimited {
                    // quota bite mid-test — acceptable
                } catch {
                    Issue.record("attestKey failed with non-quota error: \(error)")
                }
            }
        }

        func generateAssertionRoundTrip() async throws {

            purgeAttestationKeychain(keychain: keychain)
            defer { purgeAttestationKeychain(keychain: keychain) }
