    ```swift
    public func deleteKey(slot: Keyslot) throws {
        // Mirror loadPrivateKey's query shape (lines 104-121) minus kSecReturnRef.
        // SecItemDelete on a SE key removes the SecAccessControl + key reference
        // from the keychain; the SE-backed private bits become inaccessible.
        let query: [CFString: Any] = [
            kSecClass: kSecClassKey,
            kSecAttrApplicationTag: slot.applicationTag,
            kSecAttrTokenID: kSecAttrTokenIDSecureEnclave,
        ]
        let status = SecItemDelete(query as CFDictionary)
        // Idempotent contract: errSecItemNotFound is success.
        guard status == errSecSuccess || status == errSecItemNotFound else {
            throw KeyStoreError.keyDeletionFailed(status)
        }
    }
    ```
