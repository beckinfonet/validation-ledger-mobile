    ```swift
    /// Phase 4 D-07 heartbeat helper. Fire-and-forget: a failed heartbeat does NOT block
    /// role-shell render. Errors log at .error level with safe fields only (status + event name);
    /// attestationObject / attestedKeyId bytes NEVER enter Logger fields (FOUND-01 PII discipline).
    ///
    /// Invocation sites (D-07):
    ///   1. scene(_:willConnectTo:) .restored branch — cold-boot heartbeat
    ///   2. handleDidBecomeActive — warm-foreground 24h check
    ///
    /// 24h threshold: reads container.attestedKeyStore.readLastHeartbeatAt(); if nil or
    /// older than 24h, fires heartbeat. On success, updates container.session.trustTier +
    /// persists lastHeartbeatAt.
    ///
    /// Stale-session edge case (RESEARCH Pitfall 6): if the session token is server-stale,
    /// the heartbeat POST returns 401 → existing Auth401ResponseInterceptor triggers
    /// LogoutService.logout(.auth401). This is NORMAL; heartbeat serves dual duty as session
    /// validity probe.
    @MainActor
    private func performHeartbeatIfNeeded(container: AppContainer) async {
        do {
            let attestedKeyStore = AttestedKeyStore(keychain: container.keychainStore)
            // 24h threshold check
            let lastHeartbeat = try attestedKeyStore.readLastHeartbeatAt()
            let now = Date()
            if let last = lastHeartbeat,
               now.timeIntervalSince(last) < 86400 {  // 24h = 86400s
                return  // No heartbeat needed yet
            }

            // Get the existing attestedKeyId (or attest fresh if missing — shouldn't happen
            // post-register, but handle gracefully)
            let (keyId, status) = try await container.attestationService.generateKeyIfNeeded()
            guard status == .attested || status == .simulatorBypass else {
                // Can't heartbeat without a key — status already routed to softwareOnly path.
                container.logger.warn(event: .init("attestation_heartbeat_skipped_no_key"),
                                      fields: [/* safe fields: status rawValue */])
                return
            }

            // Fetch challenge (D-05)
            let challengeResponse = try await container.apiClient.send(DeviceChallengeEndpoint())
            guard let challengeData = Data(base64Encoded: challengeResponse.challenge) else {
                container.logger.error(event: .init("attestation_heartbeat_bad_challenge"),
                                       fields: [/* safe fields only */])
                return
            }

            // Generate assertion (D-06)
            let assertion = try await container.attestationService.generateAssertion(
                keyId: keyId,
                challenge: challengeData
            )

            // Read session token from Keychain (existing Phase 3 pattern)
            let sessionTokenData = try container.keychainStore.get(.sessionToken)
            guard let sessionToken = String(data: sessionTokenData, encoding: .utf8) else {
                container.logger.error(event: .init("attestation_heartbeat_no_session_token"),
                                       fields: [])
                return
            }

            // POST /device/heartbeat
            let heartbeatResponse = try await container.apiClient.send(DeviceHeartbeatEndpoint(
                sessionToken: sessionToken,
                attestedKeyId: keyId,
                assertion: assertion
            ))

            // Update trustTier (D-12) + persist lastHeartbeatAt
            container.session.trustTier = heartbeatResponse.trustTier
            try attestedKeyStore.writeLastHeartbeatAt(heartbeatResponse.heartbeatAcceptedAt)

            container.logger.info(event: .init("attestation_heartbeat_ok"),
                                  fields: [/* status + trustTier rawValue — safe */])
        } catch {
            // Silent-fail: do not block role-shell render. Backend will drive re-attest
            // on next /device/register if trustTier demands it. 401 errors are handled
            // by the existing Auth401ResponseInterceptor elsewhere.
            container.logger.error(event: .init("attestation_heartbeat_failed"),
                                   fields: [/* safe — String(describing: error) WITHOUT userInfo */])
        }
    }
    ```

    ```swift
    // validationLedger/Core/Networking/Interceptors/AttestationErrorResponseInterceptor.swift
    // Phase 4 DEV-04 / D-04 automatic path: ResponseInterceptor that intercepts 4xx responses
    // on /device/register + /device/heartbeat, inspects the error_code field, and on the three
    // canonical D-04 triggers (attestationInvalid / nonceExpired / keyCompromised):
    //   1. Calls container.attestationService.clearPersistedKeyId()  — wipe Keychain entry
    //   2. Calls container.attestationService.generateKeyIfNeeded()  — regenerate via DCAppAttestService.generateKey
    //   3. Retries the original URLRequest ONCE  — returns the retry result
    //
    // Mirrors the Phase 3 Auth401ResponseInterceptor pattern (same Core/Networking/Interceptors/
    // directory). Registered in AppContainer's interceptor chain alongside Auth401ResponseInterceptor
    // (Plan 06 wiring edit — one-line append).
    //
    // Non-attestation paths: filtered out via `eligiblePaths`. Only /device/register and
    // /device/heartbeat are eligible; a 4xx on any other path flows through unchanged.
    //
    // Non-canonical error codes: if the 4xx body's error_code is something other than the three
    // canonical triggers, flow through unchanged. No clear, no retry.
    //
    // Body-parse failure: if the 4xx response body is not JSON, treat as non-canonical and flow
    // through unchanged. Missing body, malformed body, non-UTF8 — all safe-fail.
    //
    // PII discipline (FOUND-01): logs only the error_code string + path; NEVER the response body
    // bytes or the attestedKeyId.

    import Foundation

    public struct AttestationErrorResponseInterceptor: ResponseInterceptor {

        /// Canonical D-04 re-attestation trigger codes. Must exactly match the backend contract
        /// (RESEARCH Q1 — wire-format names TBD by backend team; update in coordination if they
        /// choose different strings).
        public static let canonicalTriggerCodes: Set<String> = [
            "attestationInvalid",
            "nonceExpired",
            "keyCompromised"
        ]

        public static let defaultEligiblePaths: Set<String> = [
            "/device/register",
            "/device/heartbeat"
        ]

        private let attestationService: any AttestationService
        private let eligiblePaths: Set<String>
        private let triggerCodes: Set<String>
        private let logger: (any Logger)?

        public init(
            attestationService: any AttestationService,
            eligiblePaths: Set<String> = AttestationErrorResponseInterceptor.defaultEligiblePaths,
            triggerCodes: Set<String> = AttestationErrorResponseInterceptor.canonicalTriggerCodes,
            logger: (any Logger)? = nil
        ) {
            self.attestationService = attestationService
            self.eligiblePaths = eligiblePaths
            self.triggerCodes = triggerCodes
            self.logger = logger
        }

        public func intercept(
            send: @Sendable (URLRequest) async throws -> (Data, HTTPURLResponse),
            request: URLRequest
        ) async throws -> (Data, HTTPURLResponse) {
            let (data, response) = try await send(request)

            // Path filter — only act on /device/register + /device/heartbeat
            guard let path = request.url?.path, eligiblePaths.contains(path) else {
                return (data, response)
            }

            // Status-code filter — only 4xx qualifies for D-04 triggering. 2xx / 3xx / 5xx flow through.
            guard (400..<500).contains(response.statusCode) else {
                return (data, response)
            }

            // Parse error_code out of the body. Safe-fail on any malformed body.
            guard let errorCode = Self.extractErrorCode(from: data),
                  triggerCodes.contains(errorCode) else {
                return (data, response)
            }

            // D-04 canonical flow. Log at info (safe fields only).
            logger?.info(event: .init("attestation_backend_reattest_triggered"), fields: [/* safe: errorCode + path */])

            // 1. Clear persisted keyId (wipe Keychain entry)
            do {
                try attestationService.clearPersistedKeyId()
            } catch {
                // Safe-fail: log + flow through original response. Do NOT throw — the caller
                // still needs the original 4xx to handle normally.
                logger?.error(event: .init("attestation_clear_persisted_key_failed"), fields: [])
                return (data, response)
            }

            // 2. Regenerate keyId (will produce a fresh attestedKeyId on next Apple call)
            do {
                _ = try await attestationService.generateKeyIfNeeded()
            } catch {
                logger?.error(event: .init("attestation_regenerate_failed"), fields: [])
                return (data, response)
            }

            // 3. Retry the original request ONCE. Return whatever the retry produces — no
            //    recursion into this interceptor (ResponseInterceptor chain is linear; a second
            //    4xx from the retry flows through unchanged so the caller sees it).
            let (retriedData, retriedResponse) = try await send(request)
            return (retriedData, retriedResponse)
        }

        /// Extract top-level `error_code` from a JSON body. Returns nil on any decode failure.
        /// Accepts both snake_case (error_code) and camelCase (errorCode) as a forward-compat measure.
        private static func extractErrorCode(from data: Data) -> String? {
            guard let obj = try? JSONSerialization.jsonObject(with: data) as? [String: Any] else {
                return nil
            }
            if let code = obj["error_code"] as? String { return code }
            if let code = obj["errorCode"] as? String { return code }
            return nil
        }
    }
    ```
