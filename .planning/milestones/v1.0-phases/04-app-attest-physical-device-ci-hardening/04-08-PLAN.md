    ```swift
    // Phase 4 D-11 + D-12: conditionally wrap the tab bar controller with a limited-trust
    // banner when the backend-returned trustTier is not hardwareAttested.
    //
    // Returns:
    //   - trustTier == .hardwareAttested → self (no wrapper needed)
    //   - trustTier == .softwareOnly    → a UIViewController wrapper whose view hierarchy is
    //                                      [ tabBarController.view (pinned to safeArea bottom + sides),
    //                                        LimitedTrustBannerView (pinned to safeArea top) ]
    //
    // The wrapper becomes the new root for the .role(role) phase; AppCoordinator's presentRoot
    // swaps window.rootViewController to this wrapper.
    //
    // iPad landscape + notch: pins to safeAreaLayoutGuide per RESEARCH Pitfall 7.
    //
    // Re-wrapping: if trustTier changes from .softwareOnly → .hardwareAttested via a heartbeat
    // (D-12), the banner does NOT auto-remove this session. Next app relaunch or next role
    // transition re-evaluates. (M1 tradeoff per CLAUDE.md simplicity; M2 can add a reactive
    // update via NotificationCenter if product demands it.)

    extension UITabBarController {
        @MainActor
        public func wrapWithLimitedTrustBanner(trustTier: TrustTier) -> UIViewController {
            guard trustTier != .hardwareAttested else {
                return self
            }

            let container = UIViewController()
            container.view.backgroundColor = .systemBackground

            let banner = LimitedTrustBannerView()
            banner.translatesAutoresizingMaskIntoConstraints = false
            container.view.addSubview(banner)

            // Make self a child VC + add its view to the container
            container.addChild(self)
            self.view.translatesAutoresizingMaskIntoConstraints = false
            container.view.addSubview(self.view)
            self.didMove(toParent: container)

            NSLayoutConstraint.activate([
                // Banner pinned to safe area top (RESEARCH Pitfall 7)
                banner.topAnchor.constraint(equalTo: container.view.safeAreaLayoutGuide.topAnchor),
                banner.leadingAnchor.constraint(equalTo: container.view.leadingAnchor),
                banner.trailingAnchor.constraint(equalTo: container.view.trailingAnchor),

                // Tab bar controller's view pinned below the banner + to all other edges
                self.view.topAnchor.constraint(equalTo: banner.bottomAnchor),
                self.view.leadingAnchor.constraint(equalTo: container.view.leadingAnchor),
                self.view.trailingAnchor.constraint(equalTo: container.view.trailingAnchor),
                self.view.bottomAnchor.constraint(equalTo: container.view.bottomAnchor),
            ])

            return container
        }
    }
    ```
