```swift
let shimmerLayer: CAGradientLayer = {
    let l = CAGradientLayer()
    l.colors = [UIColor.clear.cgColor, UIColor.white.withAlphaComponent(0.25).cgColor, UIColor.clear.cgColor]
    l.startPoint = CGPoint(x: 0, y: 0.5)
    l.endPoint = CGPoint(x: 1, y: 0.5)
    l.locations = [0, 0.5, 1]
    return l
}()

public override func layoutSubviews() {
    super.layoutSubviews()
    shimmerLayer.frame = contentView.bounds
    startShimmer()
}

private func startShimmer() {
    guard shimmerLayer.animation(forKey: "shimmer") == nil else { return }
    let anim = CABasicAnimation(keyPath: "locations")
    anim.fromValue = [-1, -0.5, 0]
    anim.toValue = [1, 1.5, 2]
    anim.duration = 1.2
    anim.repeatCount = .infinity
    anim.isRemovedOnCompletion = false
    shimmerLayer.add(anim, forKey: "shimmer")
}
```
